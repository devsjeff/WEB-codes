// mental model
Think of PostgreSQL like an Excel sheet on steroids — rows, columns, strict types. But it guarantees your data is always correct (ACID). Best for: structured data with relationships — users, orders, products, anything where rows relate to each other.

table creation — template
create table
CREATE TABLE users (
  id         BIGSERIAL PRIMARY KEY,
  email      TEXT NOT NULL UNIQUE,
  username   TEXT NOT NULL,
  role       TEXT NOT NULL DEFAULT 'user',
  is_active  BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE posts (
  id      BIGSERIAL PRIMARY KEY,
  title   TEXT NOT NULL,
  content TEXT,
  user_id BIGINT REFERENCES users(id)
          ON DELETE CASCADE,
  -- user delete → posts delete
  created_at TIMESTAMPTZ DEFAULT NOW()
);
constraints — rules on data
CREATE TABLE products (
  id    BIGSERIAL PRIMARY KEY,
  name  TEXT NOT NULL,
  price NUMERIC(10,2) CHECK (price > 0),
  stock INTEGER DEFAULT 0 CHECK (stock >= 0),
  UNIQUE(name)  -- name must be unique
);

queries — full reference
SELECT patterns
-- basic
SELECT * FROM users;
SELECT id, email FROM users WHERE is_active = true;

-- filter operators
WHERE age BETWEEN 18 AND 30
WHERE name LIKE 'Dev%'    -- starts with Dev
WHERE name ILIKE '%dev%'  -- case insensitive
WHERE id IN (1, 2, 3)
WHERE email IS NOT NULL

-- sort + limit
ORDER BY created_at DESC
LIMIT 10 OFFSET 20  -- pagination: page 3

-- count, sum, avg
SELECT COUNT(*), AVG(price), SUM(amount)
FROM orders;
INSERT · UPDATE · DELETE
-- insert single
INSERT INTO users (email, username)
VALUES ('dev@gmail.com', 'Dev')
RETURNING *;  -- get inserted row back

-- insert multiple
INSERT INTO users (email, username) VALUES
  ('a@x.com', 'A'),
  ('b@x.com', 'B');

-- upsert (insert or update)
INSERT INTO users (email, username)
VALUES ('dev@x.com', 'Dev')
ON CONFLICT (email)
DO UPDATE SET username = EXCLUDED.username;

-- update
UPDATE users SET username = 'NewDev'
WHERE id = 1 RETURNING *;

-- delete
DELETE FROM users WHERE id = 1;
JOINs
-- INNER JOIN — match hona chahiye dono mein
SELECT u.username, p.title
FROM posts p
INNER JOIN users u ON p.user_id = u.id;

-- LEFT JOIN — left table sab aayega
-- right mein nahi toh NULL
SELECT u.username, COUNT(p.id) as post_count
FROM users u
LEFT JOIN posts p ON p.user_id = u.id
GROUP BY u.id, u.username;

-- multiple joins
SELECT u.username, p.title, c.body
FROM posts p
JOIN users u ON p.user_id = u.id
JOIN comments c ON c.post_id = p.id
WHERE u.is_active = true
ORDER BY p.created_at DESC;
GROUP BY · CTE · Window
-- GROUP BY + HAVING
SELECT user_id, COUNT(*) as total
FROM posts
GROUP BY user_id
HAVING COUNT(*) > 5;  -- filter on group

-- CTE (clean complex queries)
WITH active AS (
  SELECT * FROM users WHERE is_active = true
)
SELECT * FROM active WHERE role = 'admin';

-- Window Function (ranking)
SELECT username, amount,
  ROW_NUMBER() OVER (
    PARTITION BY user_id
    ORDER BY amount DESC
  ) as rank
FROM orders;
indexes — when and how
index types
-- B-Tree (default) — WHERE, ORDER BY
CREATE INDEX idx_users_email
ON users(email);

-- Composite — multiple columns
CREATE INDEX idx_posts_user_date
ON posts(user_id, created_at DESC);

-- Partial — sirf kuch rows
CREATE INDEX idx_active_users
ON users(email) WHERE is_active = true;

-- GIN — JSONB aur arrays ke liye
CREATE INDEX idx_meta
ON events USING GIN(data);

-- check if index is being used
EXPLAIN ANALYZE
SELECT * FROM users WHERE email = 'x@x.com';
index rules
-- INDEX BANAO JAB:
WHERE clause mein column  → index
JOIN condition mein column → index  
ORDER BY mein column      → index
Foreign key column        → index

-- INDEX MAT BANAO JAB:
Small table (< 1000 rows) → skip
Columns jo rarely use ho  → skip
Bahut zyada writes ho     → careful
-- index writes bhi slow karta hai

-- EXPLAIN output mein dekho:
"Seq Scan"   → index nahi — slow
"Index Scan" → index use ho raha — fast

async with pool.acquire() as conn:
    async with conn.transaction():
        # agar koi bhi fail → sab rollback

          "UPDATE acc SET bal=bal-$1 WHERE id=$2",
          1000, 1
        )
        await conn.execute(
          "UPDATE acc SET bal=bal+$1 WHERE id=$2",
          1000, 2
        )

golden rules
1
Hamesha %s ya $1 use karo — kabhi bhi f-string mein SQL mat likho → SQL injection hoga
2
Connection pool banao — har request pe naya connection mat banao → slow aur expensive
3
TIMESTAMPTZ use karo — TIMESTAMP nahi → timezone bugs production mein
4
Migrations use karo (Alembic) — direct ALTER TABLE production mein mat karo
5
EXPLAIN ANALYZE karo slow queries pe — "Seq Scan" dikhey toh index banao
6
ON DELETE CASCADE soch ke lagao — galat lagaya toh data automatically delete ho jaayega

basic commands
import redis

r = redis.Redis(host="localhost", port=6379, db=0)

# STRING — set + get
r.set("name", "Dev")
r.set("name", "Dev", ex=3600)  # expire 1hr
r.get("name")     # b"Dev"
r.delete("name")

# COUNTER
r.incr("visits")   # 1
r.incr("visits")   # 2
r.incrby("visits", 5)  # 7

# HASH — like dict
r.hset("user:1", mapping={
    "name": "Dev", "age": "25"
})
r.hget("user:1", "name")   # Dev
r.hgetall("user:1")         # full dict

# LIST — queue
r.rpush("queue", "task1")  # right push
r.lpop("queue")            # left pop
caching pattern — most common use
import json
import redis

r = redis.Redis(host="localhost")

async def get_user(user_id: int):
    cache_key = f"user:{user_id}"

    # 1. cache check karo
    cached = r.get(cache_key)
    if cached:
        return json.loads(cached)  # cache hit

    # 2. DB se lo
    user = await db.fetchrow(
        "SELECT * FROM users WHERE id=$1", user_id
    )

    # 3. cache mein save karo
    r.set(cache_key,
         json.dumps(dict(user)),
         ex=300)  # 5 min

    return user

# cache invalidate karo jab update ho
async def update_user(user_id, data):
    await db.execute("UPDATE users...")
    r.delete(f"user:{user_id}")  # cache clear
rate limiting
from fastapi import HTTPException

def rate_limit(user_id: str, limit=10, window=60):
    key = f"rate:{user_id}"
    current = r.incr(key)

    if current == 1:
        r.expire(key, window)  # first req — timer set

    if current > limit:
        raise HTTPException(429, "Too many requests")

session store + async redis
# aioredis — async redis (FastAPI)
import aioredis

redis_client = await aioredis.create_redis_pool(
    "redis://localhost"
)

# session store
import secrets

async def create_session(user_id: int):
    token = secrets.token_hex(32)
    await redis_client.setex(
        f"session:{token}",
        86400,     # 24 hours
        user_id
    )
    return token

async def get_session(token: str):
    user_id = await redis_client.get(f"session:{token}")
    return int(user_id) if user_id else None

typical startup stack
DB
PostgreSQL — primary database. Users, orders, products, sab kuch yahan.
Cache
Redis — PostgreSQL ke saath. Frequently accessed data cache karo. Sessions store karo. Rate limiting.
Files
AWS S3 — images, videos, documents, exports. PostgreSQL mein sirf S3 key store karo.
Logs
MongoDB — optional. Event logs, analytics, flexible data jo baar baar badle.
Skip
Firebase — sirf agar backend bilkul nahi likhna. Production startup mein generally avoid karo.
